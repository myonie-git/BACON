cmake_minimum_required(VERSION 3.16)
project(BACON VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)

option(BACON_ENABLE_OPENMP "Enable parallel software experiments" ON)
option(BACON_ENABLE_CUDA "Enable the historical CUDA baselines" OFF)
option(BACON_ENABLE_OMPL "Enable RRT/pRRT experiments with the pinned OMPL patch" OFF)
option(BACON_BUILD_GEOMETRIC_TESTS "Build vendored geometric_shapes tests" OFF)

find_package(PkgConfig REQUIRED)
pkg_check_modules(TINYXML2 REQUIRED IMPORTED_TARGET tinyxml2)
pkg_check_modules(FCL REQUIRED IMPORTED_TARGET fcl)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
find_package(Boost 1.71 REQUIRED COMPONENTS system filesystem)
find_package(console_bridge REQUIRED)
find_package(octomap REQUIRED)
find_package(random_numbers REQUIRED)
find_package(resource_retriever REQUIRED)

set(TEST_RESOURCES_DIR "${CMAKE_CURRENT_SOURCE_DIR}/resources")
configure_file(resources/config.h.in "${CMAKE_CURRENT_BINARY_DIR}/resources/config.h")
add_subdirectory(geometric_shapes)

add_library(cdu_impl
    src/aabb.cpp src/obb.cpp src/bvh_node_base.cpp src/bvh_tree.cpp
    src/collision_env.cpp src/collision_object.cpp src/urdf_model.cpp src/robot_node_base.cpp)
target_include_directories(cdu_impl PUBLIC
    "${CMAKE_CURRENT_SOURCE_DIR}/include" "${CMAKE_CURRENT_SOURCE_DIR}"
    "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/geometric_shapes/include"
    "${CMAKE_CURRENT_SOURCE_DIR}/geometric_shapes/test"
    ${random_numbers_INCLUDE_DIRS} ${resource_retriever_INCLUDE_DIRS})
target_compile_definitions(cdu_impl PUBLIC BACON_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(cdu_impl PUBLIC Eigen3::Eigen PkgConfig::TINYXML2 PkgConfig::FCL geometric_shapes)

if(BACON_ENABLE_OPENMP)
    find_package(OpenMP REQUIRED)
    target_link_libraries(cdu_impl PUBLIC OpenMP::OpenMP_CXX)
endif()

add_executable(cdu_cimpl src/main.cpp)
target_link_libraries(cdu_cimpl PRIVATE cdu_impl)
add_executable(bacon_smoke examples/smoke.cpp)
target_link_libraries(bacon_smoke PRIVATE cdu_impl)
include(CTest)
if(BUILD_TESTING)
    add_test(NAME bacon_models_and_copied_poses COMMAND bacon_smoke)
endif()

function(bacon_experiment name)
    add_executable(${name} EXCLUDE_FROM_ALL "test/${name}.cpp")
    target_link_libraries(${name} PRIVATE cdu_impl)
endfunction()

if(BACON_ENABLE_OMPL)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/setup_ompl.py" --check
        RESULT_VARIABLE ompl_ready)
    if(NOT ompl_ready EQUAL 0)
        message(FATAL_ERROR "Prepare the pinned OMPL dependency: python3 scripts/setup_ompl.py")
    endif()
    set(OMPL_REGISTRATION OFF CACHE BOOL "" FORCE)
    foreach(feature DEMOS TESTS PYBINDINGS PYTESTS VAMP)
        set(OMPL_BUILD_${feature} OFF CACHE BOOL "" FORCE)
    endforeach()
    add_subdirectory(third_party/ompl EXCLUDE_FROM_ALL)
    foreach(name sim_prrt_plan sim_prrt_sample_limit sim_rrt_sample_limit)
        bacon_experiment(${name})
        target_link_libraries(${name} PRIVATE ompl::ompl)
        target_compile_definitions(${name} PRIVATE CDU_CIMPL_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
    endforeach()
endif()
